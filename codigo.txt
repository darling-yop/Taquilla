function leerRuta() {
  const hash = window.location.hash.slice(1);
  if (!hash) return { tipo: "inicio" };

  const [pagina, query = ""] = hash.split("?", 2);
  if (pagina.startsWith("evento/")) {
    return { tipo: "evento", slug: decodeURIComponent(pagina.slice(7)) };
  }
  if (pagina.startsWith("categoria/")) {
    return {
      tipo: "categoria",
      categoria: decodeURIComponent(pagina.slice(10)),
    };
  }
  if (pagina === "buscar") {
    return {
      tipo: "buscar",
      filtros: Object.fromEntries(new URLSearchParams(query)),
    };
  }
   if (pagina === "explorar") {
    return {
      tipo: "explorar",
      filtros: Object.fromEntries(new URLSearchParams(query)),
    };
  }
  if (pagina === "ingresar" || pagina === "crearCuenta") {
    return { tipo: pagina };
  }
  return { tipo: "inicio" };
}

export default function GestorCookies({ abierto, onAbrir, onCerrar }) {
  const { idioma, consentimiento, guardarConsentimiento, error } =
    usePreferencias();
  const [mensaje, setMensaje] = useState(false);
  const t = interfaz[idioma];
  function guardar(permitir) {
    if (guardarConsentimiento(permitir)) {
      setMensaje(true);
      onCerrar();
    }
  }
  return (
    <>
      {!consentimiento && (
        <BannerCookies
          textos={t}
          onAceptar={() => guardar(true)}
          onRechazar={() => guardar(false)}
          onConfigurar={onAbrir}
        />
      )}
      {error && !abierto && (
        <p className="error-cookies" role="alert">
          {t.error}
        </p>
      )}
      <PreferenciasCookies
        abierto={abierto}
        textos={t}
        permitir={consentimiento === "aceptado"}
        onGuardar={guardar}
        onCerrar={onCerrar}
        error={error}
      />
      <span className="solo-lectores" role="status">
        {mensaje ? t.guardado : ""}
      </span>
    </>
  );
}

import { createContext, useContext, useEffect, useState } from "react";
import {
  leerCookie,
  guardarCookie,
  borrarCookie,
} from "../services/cookies.js";
import { proyecto } from "../config/proyecto.js";

const PreferenciasContext = createContext(null);

export function PreferenciasProvider({ children }) {
  const [consentimiento, setConsentimiento] = useState(() => {
    const valor = leerCookie("consentimiento");
    return ["aceptado", "rechazado"].includes(valor) ? valor : null;
  });
  const [idioma, setIdioma] = useState(() =>
    leerCookie("consentimiento") === "aceptado" && leerCookie("idioma") === "en"
      ? "en"
      : "es",
  );
  const [error, setError] = useState(false);
  useEffect(() => {
    document.documentElement.lang = idioma;
    document.title = `${proyecto.nombre} Â· ${idioma === "en" ? "Home" : "Inicio"}`;
  }, [idioma]);
  function cambiarIdioma(valor) {
    setIdioma(valor);
    try {
      if (consentimiento === "aceptado") guardarCookie("idioma", valor);
      setError(false);
    } catch {
      setError(true);
    }
  }
  function guardarConsentimiento(permitir) {
    try {
      const valor = permitir ? "aceptado" : "rechazado";
      guardarCookie("consentimiento", valor);
      if (permitir) guardarCookie("idioma", idioma);
      else borrarCookie("idioma");
      setConsentimiento(valor);
      setError(false);
      return true;
    } catch {
      setError(true);
      return false;
    }
  }
  return (
    <PreferenciasContext.Provider
      value={{
        idioma,
        cambiarIdioma,
        consentimiento,
        guardarConsentimiento,
        error,
      }}
    >
      {children}
    </PreferenciasContext.Provider>
  );
}

export function usePreferencias() {
  const valor = useContext(PreferenciasContext);
  if (!valor) throw new Error("PreferenciasProvider requerido");
  return valor;
}

export function useRecurso(ruta) {
  const [estado, setEstado] = useState({
    ruta: null,
    datos: null,
    error: null,
    cargando: true,
  });
  const [intento, setIntento] = useState(0);
  useEffect(() => {
    if (!ruta) return;
    const controlador = new AbortController();
    let vigente = true;
    setEstado({ ruta, datos: null, error: null, cargando: true });
    consultar(ruta, { signal: controlador.signal })
      .then((datos) => {
        if (vigente) setEstado({ ruta, datos, error: null, cargando: false });
      })
      .catch((error) => {
        if (vigente && error.name !== "AbortError")
          setEstado({ ruta, datos: null, error, cargando: false });
      });
    return () => {
      vigente = false;
      controlador.abort();
    };
  }, [ruta, intento]);
  const actual =
    estado.ruta === ruta
      ? estado
      : { datos: null, error: null, cargando: Boolean(ruta) };
  return { ...actual, recargar: () => setIntento((valor) => valor + 1) };
}

export async function consultar(ruta, { signal } = {}) {
  const respuesta = await fetch(`${proyecto.api}${ruta}`, {
    signal,
    headers: { Accept: "application/json" },
  });
  const datos = await respuesta.json();
  if (!respuesta.ok) {
    const error = new Error(datos.error?.mensaje || `HTTP ${respuesta.status}`);
    error.codigo = datos.error?.codigo;
    error.campos = datos.error?.campos;
    throw error;
  }
  return datos;
}

export function leerCookie(clave) {
  const prefijo = `${nombre(clave)}=`;
  const cookie = document.cookie
    .split("; ")
    .find((valor) => valor.startsWith(prefijo));
  try {
    return cookie ? decodeURIComponent(cookie.slice(prefijo.length)) : null;
  } catch {
    return null;
  }
}

export function guardarCookie(clave, valor) {
  document.cookie = `${nombre(clave)}=${encodeURIComponent(valor)}; Path=/; Max-Age=15552000; SameSite=Lax${location.protocol === "https:" ? "; Secure" : ""}`;
  if (leerCookie(clave) !== valor) throw new Error("cookie_no_guardada");
}

export function borrarCookie(clave) {

}

    const ultimo = new Date(Date.UTC(anio, mes, 0)).toISOString().slice(0, 10);
    return { desde: hoy, hasta: ultimo };
  }

  const diaSemana = new Date(`${hoy}T12:00:00Z`).getUTCDay();
  if (diaSemana === 0) return { desde: hoy, hasta: hoy };

}
